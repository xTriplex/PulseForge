cmake_minimum_required(VERSION 3.24)
project(PulseForgeRuntimeTargetClosureFixture NONE)

include("${PULSEFORGE_SOURCE_ROOT}/CMake/PulseForgeRuntimeTargetClosure.cmake")

add_library(PulseForgeAssetImporters INTERFACE)
add_library(AuthoringAlias ALIAS PulseForgeAssetImporters)
add_library(IndirectAuthoring INTERFACE)
add_library(RuntimeFoundation INTERFACE)
add_library(CookedExecutable INTERFACE)
add_library(ConfigImportedArchive STATIC IMPORTED GLOBAL)
add_library(ModernImportedArchive STATIC IMPORTED GLOBAL)
set_property(TARGET ConfigImportedArchive PROPERTY IMPORTED_CONFIGURATIONS "DEBUG;RELEASE;NOCONFIG")
set_property(TARGET ConfigImportedArchive PROPERTY IMPORTED_LOCATION_DEBUG "${CMAKE_CURRENT_BINARY_DIR}/fixture-debug.lib")
set_property(TARGET ConfigImportedArchive PROPERTY IMPORTED_LOCATION_RELEASE "${CMAKE_CURRENT_BINARY_DIR}/fixture-release.lib")
set_property(TARGET ConfigImportedArchive PROPERTY IMPORTED_LINK_INTERFACE_LIBRARIES_DEBUG AuthoringAlias)
set_property(TARGET ConfigImportedArchive PROPERTY IMPORTED_LINK_INTERFACE_LIBRARIES_NOCONFIG AuthoringAlias)
set_property(TARGET ModernImportedArchive PROPERTY IMPORTED_CONFIGURATIONS "DEBUG")
set_property(TARGET ModernImportedArchive PROPERTY IMPORTED_LOCATION_DEBUG "${CMAKE_CURRENT_BINARY_DIR}/modern-fixture-debug.lib")
set_property(TARGET ModernImportedArchive PROPERTY IMPORTED_LINK_INTERFACE_LIBRARIES_DEBUG AuthoringAlias)
add_library(LoopA INTERFACE)
add_library(LoopB INTERFACE)

target_link_libraries(LoopA INTERFACE LoopB)
target_link_libraries(LoopB INTERFACE LoopA)

target_link_libraries(IndirectAuthoring INTERFACE "$<$<CONFIG:Debug>:AuthoringAlias>")
target_link_libraries(RuntimeFoundation INTERFACE IndirectAuthoring LoopA)
target_link_libraries(CookedExecutable INTERFACE RuntimeFoundation ConfigImportedArchive ModernImportedArchive
	"$<LINK_ONLY:Threads::Threads>")

if(PULSEFORGE_EXPECT_FORBIDDEN)
	pulseforge_assert_no_runtime_target_dependency(CookedExecutable PulseForgeAssetImporters)
elseif(PULSEFORGE_EXPECT_NOCONFIG_FORBIDDEN)
	add_library(Threads::Threads INTERFACE IMPORTED)
	set_property(TARGET ModernImportedArchive PROPERTY INTERFACE_LINK_LIBRARIES Threads::Threads)
	set_property(TARGET CookedExecutable PROPERTY INTERFACE_LINK_LIBRARIES
		"RuntimeFoundation;ConfigImportedArchive;ModernImportedArchive;Threads::Threads")
	set_property(TARGET IndirectAuthoring PROPERTY INTERFACE_LINK_LIBRARIES "")
	set_property(TARGET ConfigImportedArchive PROPERTY IMPORTED_LINK_INTERFACE_LIBRARIES_DEBUG "")
	pulseforge_assert_no_runtime_target_dependency(CookedExecutable PulseForgeAssetImporters)
else()
	add_library(Threads::Threads INTERFACE IMPORTED)
	set_property(TARGET ModernImportedArchive PROPERTY INTERFACE_LINK_LIBRARIES Threads::Threads)
	set_property(TARGET CookedExecutable PROPERTY INTERFACE_LINK_LIBRARIES
		"RuntimeFoundation;ConfigImportedArchive;ModernImportedArchive;Threads::Threads")
	set_property(TARGET IndirectAuthoring PROPERTY INTERFACE_LINK_LIBRARIES "")
	set_property(TARGET ConfigImportedArchive PROPERTY IMPORTED_LINK_INTERFACE_LIBRARIES_DEBUG "")
	set_property(TARGET ConfigImportedArchive PROPERTY IMPORTED_LINK_INTERFACE_LIBRARIES_NOCONFIG "")
	pulseforge_assert_no_runtime_target_dependency(CookedExecutable PulseForgeAssetImporters)
endif()
